name: Demo authentic producer isolated review
on:
  pull_request:
    types: [opened, synchronize, reopened]
    branches: [main]
    paths:
      - 'scripts/ci/demo-authentic-review/**'
      - '.github/workflows/demo-authentic-producer-review.yml'
permissions:
  contents: read
concurrency:
  group: demo-authentic-review-${{ github.event.pull_request.number }}
  cancel-in-progress: false
jobs:
  synthetic-review:
    if: github.repository == 'jfcaicedoobando-arch/elogistix' && github.event.pull_request.draft == true && github.event.pull_request.head.repo.full_name == github.repository && github.event.pull_request.head.ref == 'fix/demo-authentic-producer-review-20261015'
    runs-on: ubuntu-24.04
    timeout-minutes: 25
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          ref: ${{ github.event.pull_request.head.sha }}
          persist-credentials: false
      - name: Offline isolated harness contracts
        run: python3 -m unittest discover -s scripts/ci/demo-authentic-review -p 'test_*.py' -v
      - name: Authentic producer and registry PostgreSQL 17.6
        env:
          DEMO_ACCEPTANCE_ISOLATED: '1'
        run: python3 -I scripts/ci/demo-authentic-review/run_ci.py --execute-disposable --output "$RUNNER_TEMP/demo-authentic-evidence"
      - name: Preserve synthetic evidence
        if: always()
        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        with:
          name: demo-authentic-review-${{ github.run_id }}-${{ github.run_attempt }}
          path: ${{ runner.temp }}/demo-authentic-evidence
          if-no-files-found: warn
          retention-days: 14
